    75→    75→    75→function VideoPlayer({
    76→    76→    76→  stream,
    77→    77→    77→  muted = false,
    78→    78→    78→  label,
    79→    79→    79→  mirrored = false,
    80→    80→    80→  className = '',
    81→    81→    81→  isProtected = false,
    82→    82→    82→}: {
    83→    83→    83→  stream: MediaStream | null
    84→    84→    84→  muted?: boolean
    85→    85→    85→  label: string
    86→    86→    86→  mirrored?: boolean

    90→    90→    90→  const videoRef = useRef<HTMLVideoElement>(null)
    91→    91→    91→
    92→    92→    92→  useEffect(() => {
    93→    93→    93→    if (videoRef.current && stream) {
    94→    94→    94→      videoRef.current.srcObject = stream
    95→    95→    95→    }
    96→    96→    96→    return () => {
    97→    97→    97→      if (videoRef.current) videoRef.current.srcObject = null
    98→    98→    98→    }
    99→    99→    99→  }, [stream])
   100→   100→   100→
   101→   101→   101→  return (
   102→   102→   102→    <div
   103→   103→   103→      className={`relative rounded-2xl overflow-hidden bg-neutral-900 ${isProtected ? 'select-none pointer-events-none' : ''} ${className}`}

   105→   105→   105→    >
   106→   106→   106→      {isProtected && <div className="absolute inset-0 z-10" aria-hidden="true" />}
   107→   107→   107→      <video
   108→   108→   108→        ref={videoRef}
   109→   109→   109→        autoPlay
   110→   110→   110→        playsInline
   111→   111→   111→        muted={muted}
   112→   112→   112→        disablePictureInPicture
   113→   113→   113→        disableRemotePlayback
   114→   114→   114→        controlsList="nodownload nofullscreen noremoteplayback"
   115→   115→   115→        className={`w-full h-full object-cover ${mirrored ? 'scale-x-[-1]' : ''}`}
   116→   116→   116→      />
   117→   117→   117→      {!stream && (
   118→   118→   118→        <div className="absolute inset-0 flex items-center justify-center text-neutral-500">

   124→   124→   124→          {label}
   125→   125→   125→        </Badge>
   126→   126→   126→      </div>
   127→   127→   127→    </div>
   128→   128→   128→  )

   135→   135→   135→function useWebRTC() {
   136→   136→   136→  const socketRef = useRef<Socket | null>(null)
   137→   137→   137→  const pcRef = useRef<RTCPeerConnection | null>(null)
   138→   138→   138→  const localStreamRef = useRef<MediaStream | null>(null)
   139→   139→   139→  const remoteStreamRef = useRef<MediaStream | null>(null)
   140→   140→   140→  const remotePeerIdRef = useRef<string | null>(null)
   141→   141→   141→
   142→   142→   142→  const [peers, setPeers] = useState<Peer[]>([])
   143→   143→   143→  const [myId, setMyId] = useState<string | null>(null)
   144→   144→   144→  const [myUsername, setMyUsername] = useState('')
   145→   145→   145→  const [isRegistered, setIsRegistered] = useState(false)
   146→   146→   146→  const [incomingCall, setIncomingCall] = useState<{ fromId: string; fromName: string; offer: RTCSessionDescriptionInit } | null>(null)
   147→   147→   147→  const [isInCall, setIsInCall] = useState(false)
   148→   148→   148→  const [remoteStream, setRemoteStream] = useState<MediaStream | null>(null)
   149→   149→   149→  const [localStream, setLocalStream] = useState<MediaStream | null>(null)
   150→   150→   150→  const [isMuted, setIsMuted] = useState(false)
   151→   151→   151→  const [isCameraOff, setIsCameraOff] = useState(false)
   152→   152→   152→  const [callStatus, setCallStatus] = useState<'idle' | 'ringing' | 'connecting' | 'connected' | 'ended'>('idle')
   153→   153→   153→
   154→   154→   154→  const myIdRef = useRef<string | null>(null)
   155→   155→   155→
   156→   156→   156→  const cleanupCall = useCallback(() => {
   157→   157→   157→    if (pcRef.current) { pcRef.current.close(); pcRef.current = null }
   158→   158→   158→    if (localStreamRef.current) { localStreamRef.current.getTracks().forEach((t) => t.stop()); localStreamRef.current = null }
   159→   159→   159→    remoteStreamRef.current = null
   160→   160→   160→    remotePeerIdRef.current = null
   161→   161→   161→    setRemoteStream(null); setLocalStream(null); setIsInCall(false)
   162→   162→   162→    setIsMuted(false); setIsCameraOff(false); setIncomingCall(null)
   163→   163→   163→  }, [])
   164→   164→   164→
   165→   165→   165→  useEffect(() => {
   166→   166→   166→    const socket = io('/?XTransformPort=3003', { transports: ['websocket', 'polling'] })
   167→   167→   167→    socketRef.current = socket
   168→   168→   168→
   169→   169→   169→    socket.on('registered', (data: { id: string; username: string }) => {
   170→   170→   170→      myIdRef.current = data.id
   171→   171→   171→      setMyId(data.id); setMyUsername(data.username); setIsRegistered(true)
   172→   172→   172→    })
   173→   173→   173→
   174→   174→   174→    socket.on('peer-list', (data: { peers: Peer[] }) => {
   175→   175→   175→      setPeers(data.peers.filter((p) => p.id !== myIdRef.current && !p.isAdmin))
   176→   176→   176→    })
   177→   177→   177→
   178→   178→   178→    socket.on('incoming-call', (data) => {
   179→   179→   179→      setIncomingCall(data); setCallStatus('ringing')
   180→   180→   180→    })
   181→   181→   181→
   182→   182→   182→    socket.on('call-answered', async (data) => {
   183→   183→   183→ const pc = pcRef.current
   184→   184→   184→      if (pc) { await pc.setRemoteDescription(new RTCSessionDescription(data.answer)); setCallStatus('connecting') }
   185→   185→   185→    })
   186→   186→   186→
   187→   187→   187→    socket.on('ice-candidate', async (data) => {
   188→   188→   188→      const pc = pcRef.current
   189→   189→   189→      if (pc && data.candidate) {
   190→   190→   190→        try { await pc.addIceCandidate(new RTCIceCandidate(data.candidate)) } catch (e) { /* ignore */ }
   191→   191→   191→      }
   192→   192→   192→    })
   193→   193→   193→
   194→   194→   194→    socket.on('call-rejected', () => { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) })
   195→   195→   195→    socket.on('call-ended', () => { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) })
   196→   196→   196→    socket.on('force-disconnected', () => { cleanupCall(); setCallStatus('ended'); setIsRegistered(false) })
   197→   197→   197→
   198→   198→   198→    return () => { socket.disconnect() }
   199→   199→   199→  }, [])
   200→   200→   200→
   201→   201→   201→  const getLocalStream = useCallback(async () => {
   202→   202→   202→    const stream = await navigator.mediaDevices.getUserMedia({ video: { width: 1280, height: 720, facingMode: 'user' }, audio: true })
   203→   203→   203→    localStreamRef.current = stream; setLocalStream(stream); return stream
   204→   204→   204→  }, [])
   205→   205→   205→
   206→   206→   206→  const createPeerConnection = useCallback((stream: MediaStream, remotePeerId: string) => {
   207→   207→   207→    const pc = new RTCPeerConnection({
   208→   208→   208→      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }, { urls: 'stun:stun1.l.google.com:19302' }]
   209→   209→   209→    })
   210→   210→   210→    pcRef.current = pc; remotePeerIdRef.current = remotePeerId
   211→   211→   211→    stream.getTracks().forEach((t) => pc.addTrack(t, stream))
   212→   212→   212→    const remoteMediaStream = new MediaStream()
   213→   213→   213→    remoteStreamRef.current = remoteMediaStream
   214→   214→   214→    pc.ontrack = (e) => { e.streams[0].getTracks().forEach((t) => remoteMediaStream.addTrack(t)); setRemoteStream(new MediaStream(remoteMediaStream.getTracks())) }
   215→   215→   215→    pc.onicecandidate = (e) => { if (e.candidate && socketRef.current) socketRef.current.emit('ice-candidate', { targetId: remotePeerId, candidate: e.candidate.toJSON() }) }
   216→   216→   216→    pc.onconnectionstatechange = () => {
   217→   217→   217→      if (pc.connectionState === 'connected') { setCallStatus('connected'); setIsInCall(true) }
   218→   218→   218→      else if (['disconnected', 'failed', 'closed'].includes(pc.connectionState)) { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) }
   219→   219→   219→    }
   220→   220→   220→    return pc
   221→   221→   221→  }, [cleanupCall, getLocalStream])
   222→   222→   222→
   223→   223→   223→  const register = useCallback((username: string) => { socketRef.current?.emit('register', { username, device: 'web-admin' }) }, [])
   224→   224→   224→
   225→   225→   225→  const callPeer = useCallback(async (targetId: string) => {
   226→   226→   226→    const stream = await getLocalStream()
   227→   227→   227→    createPeerConnection(stream, targetId)
   228→   228→   228→    const pc = pcRef.current; if (!pc) return
   229→   229→   229→    const offer = await pc.createOffer(); await pc.setLocalDescription(offer)
   230→   230→   230→    socketRef.current?.emit('call-offer', { targetId, offer: offer.toJSON() }); setCallStatus('connecting')
   231→   231→   231→  }, [getLocalStream, createPeerConnection])
   232→   232→   232→
   233→   233→   233→  const acceptCall = useCallback(async () => {
   234→   234→   234→    if (!incomingCall) return
   235→   235→   235→    const stream = await getLocalStream()
   236→   236→   236→    createPeerConnection(stream, incomingCall.fromId)
   237→   237→   237→    const pc = pcRef.current; if (!pc) return
   238→   238→   238→    await pc.setRemoteDescription(new RTCSessionDescription(incomingCall.offer))
   239→   239→   239→    const answer = await pc.createAnswer(); await pc.setLocalDescription(answer)
   240→   240→   240→    socketRef.current?.emit('call-answer', { targetId: incomingCall.fromId, answer: answer.toJSON() })
   241→   241→   241→    setIncomingCall(null); setCallStatus('connecting')
   242→   242→   242→  }, [incomingCall, getLocalStream, createPeerConnection])
   243→   243→   243→
   244→   244→   244→  const rejectCall = useCallback(() => {
   245→   245→   245→    if (incomingCall && socketRef.current) socketRef.current.emit('call-rejected', { targetId: incomingCall.fromId })
   246→   246→   246→    setIncomingCall(null); setCallStatus('idle')
   247→   247→   247→  }, [incomingCall])
   248→   248→   248→
   249→   249→   249→  const endCall = useCallback(() => {
   250→   250→   250→    if (socketRef.current && remotePeerIdRef.current) socketRef.current.emit('call-ended', { targetId: remotePeerIdRef.current })
   251→   251→   251→    cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000)
   252→   252→   252→  }, [cleanupCall])
   253→   253→   253→
   254→   254→   254→  const toggleMute = useCallback(() => { localStreamRef.current?.getAudioTracks().forEach((t) => { t.enabled = !t.enabled }); setIsMuted((p) => !p) }, [])
   255→   255→   255→  const toggleCamera = useCallback(() => { localStreamRef.current?.getVideoTracks().forEach((t) => { t.enabled = !t.enabled }); setIsCameraOff((p) => !p) }, [])
   256→   256→   256→
   257→   257→   257→  return { peers, myId, myUsername, isRegistered, register, callPeer, incomingCall, acceptCall, rejectCall, isInCall, remoteStream, localStream, endCall, isMuted, isCameraOff, toggleMute, toggleCamera, callStatus, socketRef }
   258→   258→   258→}

   348→   348→   348→          <Card className="bg-neutral-900/60 border-neutral-800">
   349→   349→   349→            <CardContent className="p-12 text-center">
   350→   350→   350→              <div className="w-10 h-10 border-3 border-emerald-400 border-t-transparent rounded-full animate-spin mx-auto mb-4" />
