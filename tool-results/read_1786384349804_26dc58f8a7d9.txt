    75→function VideoPlayer({
    76→  stream,
    77→  muted = false,
    78→  label,
    79→  mirrored = false,
    80→  className = '',
    81→  isProtected = false,
    82→}: {
    83→  stream: MediaStream | null
    84→  muted?: boolean
    85→  label: string
    86→  mirrored?: boolean
    87→  className?: string
    88→  isProtected?: boolean
    89→}) {
    90→  const videoRef = useRef<HTMLVideoElement>(null)
    91→
    92→  useEffect(() => {
    93→    if (videoRef.current && stream) {
    94→      videoRef.current.srcObject = stream
    95→    }
    96→    return () => {
    97→      if (videoRef.current) videoRef.current.srcObject = null
    98→    }
    99→  }, [stream])
   100→
   101→  return (
   102→    <div
   103→      className={`relative rounded-2xl overflow-hidden bg-neutral-900 ${isProtected ? 'select-none pointer-events-none' : ''} ${className}`}
   104→      onContextMenu={(e) => isProtected && e.preventDefault()}
   105→    >
   106→      {isProtected && <div className="absolute inset-0 z-10" aria-hidden="true" />}
   107→      <video
   108→        ref={videoRef}
   109→        autoPlay
   110→        playsInline
   111→        muted={muted}
   112→        disablePictureInPicture
   113→        disableRemotePlayback
   114→        controlsList="nodownload nofullscreen noremoteplayback"
   115→        className={`w-full h-full object-cover ${mirrored ? 'scale-x-[-1]' : ''}`}
   116→      />
   117→      {!stream && (
   118→        <div className="absolute inset-0 flex items-center justify-center text-neutral-500">
   119→          <Video className="w-16 h-16 opacity-30" />
   120→        </div>
   121→      )}
   122→      <div className="absolute bottom-3 left-3">
   123→        <Badge variant="secondary" className="bg-black/60 text-white border-0 backdrop-blur-sm text-sm">
   124→          {label}
   125→        </Badge>
   126→      </div>
   127→    </div>
   128→  )
   129→}

   135→function useWebRTC() {
   136→  const socketRef = useRef<Socket | null>(null)
   137→  const pcRef = useRef<RTCPeerConnection | null>(null)
   138→  const localStreamRef = useRef<MediaStream | null>(null)
   139→  const remoteStreamRef = useRef<MediaStream | null>(null)
   140→  const remotePeerIdRef = useRef<string | null>(null)
   141→
   142→  const [peers, setPeers] = useState<Peer[]>([])
   143→  const [myId, setMyId] = useState<string | null>(null)
   144→  const [myUsername, setMyUsername] = useState('')
   145→  const [isRegistered, setIsRegistered] = useState(false)
   146→  const [incomingCall, setIncomingCall] = useState<{ fromId: string; fromName: string; offer: RTCSessionDescriptionInit } | null>(null)
   147→  const [isInCall, setIsInCall] = useState(false)
   148→  const [remoteStream, setRemoteStream] = useState<MediaStream | null>(null)
   149→  const [localStream, setLocalStream] = useState<MediaStream | null>(null)
   150→  const [isMuted, setIsMuted] = useState(false)
   151→  const [isCameraOff, setIsCameraOff] = useState(false)
   152→  const [callStatus, setCallStatus] = useState<'idle' | 'ringing' | 'connecting' | 'connected' | 'ended'>('idle')
   153→
   154→  const myIdRef = useRef<string | null>(null)
   155→
   156→  const cleanupCall = useCallback(() => {
   157→    if (pcRef.current) { pcRef.current.close(); pcRef.current = null }
   158→    if (localStreamRef.current) { localStreamRef.current.getTracks().forEach((t) => t.stop()); localStreamRef.current = null }
   159→    remoteStreamRef.current = null
   160→    remotePeerIdRef.current = null
   161→    setRemoteStream(null); setLocalStream(null); setIsInCall(false)
   162→    setIsMuted(false); setIsCameraOff(false); setIncomingCall(null)
   163→  }, [])
   164→
   165→  useEffect(() => {
   166→    const socket = io('/?XTransformPort=3003', { transports: ['websocket', 'polling'] })
   167→    socketRef.current = socket
   168→
   169→    socket.on('registered', (data: { id: string; username: string }) => {
   170→      myIdRef.current = data.id
   171→      setMyId(data.id); setMyUsername(data.username); setIsRegistered(true)
   172→    })
   173→
   174→    socket.on('peer-list', (data: { peers: Peer[] }) => {
   175→      setPeers(data.peers.filter((p) => p.id !== myIdRef.current && !p.isAdmin))
   176→    })
   177→
   178→    socket.on('incoming-call', (data) => {
   179→      setIncomingCall(data); setCallStatus('ringing')
   180→    })
   181→
   182→    socket.on('call-answered', async (data) => {
   183→ const pc = pcRef.current
   184→      if (pc) { await pc.setRemoteDescription(new RTCSessionDescription(data.answer)); setCallStatus('connecting') }
   185→    })
   186→
   187→    socket.on('ice-candidate', async (data) => {
   188→      const pc = pcRef.current
   189→      if (pc && data.candidate) {
   190→        try { await pc.addIceCandidate(new RTCIceCandidate(data.candidate)) } catch (e) { /* ignore */ }
   191→      }
   192→    })
   193→
   194→    socket.on('call-rejected', () => { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) })
   195→    socket.on('call-ended', () => { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) })
   196→    socket.on('force-disconnected', () => { cleanupCall(); setCallStatus('ended'); setIsRegistered(false) })
   197→
   198→    return () => { socket.disconnect() }
   199→  }, [])
   200→
   201→  const getLocalStream = useCallback(async () => {
   202→    const stream = await navigator.mediaDevices.getUserMedia({ video: { width: 1280, height: 720, facingMode: 'user' }, audio: true })
   203→    localStreamRef.current = stream; setLocalStream(stream); return stream
   204→  }, [])
   205→
   206→  const createPeerConnection = useCallback((stream: MediaStream, remotePeerId: string) => {
   207→    const pc = new RTCPeerConnection({
   208→      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }, { urls: 'stun:stun1.l.google.com:19302' }]
   209→    })
   210→    pcRef.current = pc; remotePeerIdRef.current = remotePeerId
   211→    stream.getTracks().forEach((t) => pc.addTrack(t, stream))
   212→    const remoteMediaStream = new MediaStream()
   213→    remoteStreamRef.current = remoteMediaStream
   214→    pc.ontrack = (e) => { e.streams[0].getTracks().forEach((t) => remoteMediaStream.addTrack(t)); setRemoteStream(new MediaStream(remoteMediaStream.getTracks())) }
   215→    pc.onicecandidate = (e) => { if (e.candidate && socketRef.current) socketRef.current.emit('ice-candidate', { targetId: remotePeerId, candidate: e.candidate.toJSON() }) }
   216→    pc.onconnectionstatechange = () => {
   217→      if (pc.connectionState === 'connected') { setCallStatus('connected'); setIsInCall(true) }
   218→      else if (['disconnected', 'failed', 'closed'].includes(pc.connectionState)) { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) }
   219→    }
   220→    return pc
   221→  }, [cleanupCall, getLocalStream])
   222→
   223→  const register = useCallback((username: string) => { socketRef.current?.emit('register', { username, device: 'web-admin' }) }, [])
   224→
   225→  const callPeer = useCallback(async (targetId: string) => {
   226→    const stream = await getLocalStream()
   227→    createPeerConnection(stream, targetId)
   228→    const pc = pcRef.current; if (!pc) return
   229→    const offer = await pc.createOffer(); await pc.setLocalDescription(offer)
   230→    socketRef.current?.emit('call-offer', { targetId, offer: offer.toJSON() }); setCallStatus('connecting')
   231→  }, [getLocalStream, createPeerConnection])
   232→
   233→  const acceptCall = useCallback(async () => {
   234→    if (!incomingCall) return
   235→    const stream = await getLocalStream()
   236→    createPeerConnection(stream, incomingCall.fromId)
   237→    const pc = pcRef.current; if (!pc) return
   238→    await pc.setRemoteDescription(new RTCSessionDescription(incomingCall.offer))
   239→    const answer = await pc.createAnswer(); await pc.setLocalDescription(answer)
   240→    socketRef.current?.emit('call-answer', { targetId: incomingCall.fromId, answer: answer.toJSON() })
   241→    setIncomingCall(null); setCallStatus('connecting')
   242→  }, [incomingCall, getLocalStream, createPeerConnection])
   243→
   244→  const rejectCall = useCallback(() => {
   245→    if (incomingCall && socketRef.current) socketRef.current.emit('call-rejected', { targetId: incomingCall.fromId })
   246→    setIncomingCall(null); setCallStatus('idle')
   247→  }, [incomingCall])
   248→
   249→  const endCall = useCallback(() => {
   250→    if (socketRef.current && remotePeerIdRef.current) socketRef.current.emit('call-ended', { targetId: remotePeerIdRef.current })
   251→    cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000)
   252→  }, [cleanupCall])
   253→
   254→  const toggleMute = useCallback(() => { localStreamRef.current?.getAudioTracks().forEach((t) => { t.enabled = !t.enabled }); setIsMuted((p) => !p) }, [])
   255→  const toggleCamera = useCallback(() => { localStreamRef.current?.getVideoTracks().forEach((t) => { t.enabled = !t.enabled }); setIsCameraOff((p) => !p) }, [])
   256→
   257→  return { peers, myId, myUsername, isRegistered, register, callPeer, incomingCall, acceptCall, rejectCall, isInCall, remoteStream, localStream, endCall, isMuted, isCameraOff, toggleMute, toggleCamera, callStatus, socketRef }
   258→}

   264→function AdminDashboard({ onLogout }: { onLogout: () => void }) {
   265→  const [snapshot, setSnapshot] = useState<AdminSnapshot | null>(null)
   266→  const [connecting, setConnecting] = useState(true)
   267→  const socketRef = useRef<Socket | null>(null)
   268→
   269→  useEffect(() => {
   270→    const socket = io('/?XTransformPort=3003', { transports: ['websocket', 'polling'] })
   271→    socketRef.current = socket
   272→
   273→    socket.on('connect', () => {
   274→      setConnecting(false)
   275→      socket.emit('admin-register', { secret: 'admin2024', username: 'Admin' })
   276→    })
   277→
   278→    socket.on('admin-registered', () => {
   279→      console.log('Admin registered')
   280→    })
   281→
   282→    socket.on('admin-snapshot', (data: AdminSnapshot) => {
   283→      setSnapshot(data)
   284→    })
   285→
   286→    socket.on('admin-rejected', (data) => {
   287→      console.error('Admin rejected:', data.reason)
   288→      setConnecting(false)
   289→    })
   290→
   291→    socket.on('disconnect', () => { setConnecting(true) })
   292→
   293→    return () => { socket.disconnect() }
   294→  }, [])
   295→
   296→  const forceDisconnect = (targetId: string) => {
   297→    socketRef.current?.emit('admin-force-disconnect', { targetId })
   298→  }
   299→
   300→  const endCall = (targetId: string) => {
   301→    socketRef.current?.emit('admin-end-call', { targetId })
   302→  }
   303→
   304→  const banUser = (targetId: string) => {
   305→    socketRef.current?.emit('admin-ban', { targetId })
   306→  }
   307→
   308→  const formatUptime = (ms: number) => {
   309→    const s = Math.floor(ms / 1000)
   310→    if (s < 60) return `${s}s`
   311→    const m = Math.floor(s / 60)
   312→    if (m < 60) return `${m}m ${s % 60}s`
   313→    const h = Math.floor(m / 60)
   314→    return `${h}h ${m % 60}m`
   315→  }
   316→
   317→  const activeCallDuration = (startedAt: number) => {
   318→    const diff = Date.now() - startedAt
   319→    const m = Math.floor(diff / 60000)
   320→    const s = Math.floor((diff % 60000) / 1000)
   321→    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   322→  }
   323→
   324→  const activeCallers = new Set(snapshot?.activeCalls.flatMap((c) => [c.callerId, c.calleeId]) ?? [])
   325→  const regularPeers = snapshot?.peers.filter((p) => !p.isAdmin) ?? []
   326→
   327→  return (
   328→    <div className="min-h-screen bg-gradient-to-br from-neutral-950 via-neutral-900 to-neutral-950">
   329→      <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   330→        {/* Header */}
   331→        <header className="flex items-center justify-between">
   332→          <div className="flex items-center gap-3">
   333→            <div className="w-10 h-10 rounded-xl bg-emerald-500/10 flex items-center justify-center">
   334→              <Shield className="w-5 h-5 text-emerald-400" />
   335→            </div>
   336→            <div>
   337→              <h1 className="text-lg font-bold text-white tracking-tight">Admin Dashboard</h1>
   338→              <p className="text-xs text-neutral-500">VideoCall control center</p>
   339→            </div>
   340→          </div>
   341→          <Button variant="ghost" onClick={onLogout} className="text-neutral-400 hover:text-red-400 gap-2">
   342→            <LogOut className="w-4 h-4" />
   343→            <span className="text-sm">Logout</span>
   344→          </Button>
   345→        </header>
   346→
   347→        {connecting && !snapshot && (
   348→          <Card className="bg-neutral-900/60 border-neutral-800">
   349→            <CardContent className="p-12 text-center">
   350→              <div className="w-10 h-10 border-3 border-emerald-400 border-t-transparent rounded-full animate-spin mx-auto mb-4" />
   351→              <p className="text-neutral-400">Connecting to server...</p>
   352→            </CardContent>
   353→          </Card>
   354→        )}
   355→
   356→        {snapshot && (
   357→          <>
   358→            {/* Stats row */}
   359→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   360→              <StatCard icon={<Users className="w-4 h-4" />} label="Online Users" value={snapshot.totalConnected} color="emerald" />
   361→              <StatCard icon={<PhoneCall className="w-4 h-4" />} label="Active Calls" value={snapshot.activeCalls.length} color="blue" />
   362→              <StatCard icon={<Smartphone className="w-4 h-4" />} label="Android" value={regularPeers.filter((p) => p.device?.includes('android')).length} color="violet" />
   363→              <StatCard icon={<Ban className="w-4 h-4" />} label="Banned" value={snapshot.bannedCount} color="red" />
   364→            </div>
   365→
   366→            {/* Active Calls */}
   367→            {snapshot.activeCalls.length > 0 && (
   368→              <div>
   369→                <h2 className="text-sm font-medium text-neutral-400 mb-3 flex items-center gap-2">
   370→                  <PhoneCall className="w-4 h-4" />
   371→                  Active Calls ({snapshot.activeCalls.length})
   372→                </h2>
   373→                <div className="grid gap-3">
   374→                  {snapshot.activeCalls.map((call) => (
   375→                    <Card key={call.id} className="bg-neutral-900/60 border-emerald-500/20">
   376→                      <CardContent className="p-4">
   377→                        <div className="flex items-center justify-between">
   378→                          <div className="flex items-center gap-3">
   379→                            <div className="flex flex-col items-center gap-1">
   380→                              <div className="w-8 h-8 rounded-full bg-emerald-500/20 flex items-center justify-center text-emerald-400 text-xs font-bold">
   381→                                {call.callerName.charAt(0).toUpperCase()}
   382→                              </div>
   383→                            </div>
   384→                            <div className="flex items-center gap-2">
   385→                              <span className="text-white text-sm font-medium">{call.callerName}</span>
   386→                              <div className="w-6 h-px bg-emerald-500" />
   387→                              <span className="text-white text-sm font-medium">{call.calleeName}</span>
   388→                            </div>
   389→                          </div>
   390→                          <div className="flex items-center gap-3">
   391→                            <div className="flex items-center gap-1.5 text-emerald-400 text-xs font-mono">
   392→                              <Clock className="w-3 h-3" />
   393→                              {activeCallDuration(call.startedAt)}

   395→                            <Button
