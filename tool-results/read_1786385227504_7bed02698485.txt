    78→function VideoPlayer({
    79→  stream,
    80→  muted = false,
    81→  label,
    82→  mirrored = false,
    83→  className = '',
    84→  isProtected = false,
    85→}: {
    86→  stream: MediaStream | null
    87→  muted?: boolean
    88→  label: string
    89→  mirrored?: boolean
    90→  className?: string
    91→  isProtected?: boolean
    92→}) {
    93→  const videoRef = useRef<HTMLVideoElement>(null)
    94→
    95→  useEffect(() => {
    96→    if (videoRef.current && stream) {
    97→      videoRef.current.srcObject = stream
    98→    }
    99→    return () => {
   100→      if (videoRef.current) videoRef.current.srcObject = null
   101→    }
   102→  }, [stream])
   103→
   104→  return (
   105→    <div
   106→      className={`relative rounded-2xl overflow-hidden bg-neutral-900 ${isProtected ? 'select-none pointer-events-none' : ''} ${className}`}
   107→      onContextMenu={(e) => isProtected && e.preventDefault()}
   108→    >
   109→      {isProtected && <div className="absolute inset-0 z-10" aria-hidden="true" />}
   110→      <video
   111→        ref={videoRef}
   112→        autoPlay
   113→        playsInline
   114→        muted={muted}
   115→        disablePictureInPicture
   116→        disableRemotePlayback
   117→        controlsList="nodownload nofullscreen noremoteplayback"
   118→        className={`w-full h-full object-cover ${mirrored ? 'scale-x-[-1]' : ''}`}
   119→      />
   120→      {!stream && (
   121→        <div className="absolute inset-0 flex items-center justify-center text-neutral-500">
   122→          <Video className="w-16 h-16 opacity-30" />
   123→        </div>
   124→      )}
   125→      <div className="absolute bottom-3 left-3">
   126→        <Badge variant="secondary" className="bg-black/60 text-white border-0 backdrop-blur-sm text-sm">
   127→          {label}
   128→        </Badge>
   129→      </div>
   130→    </div>
   131→  )
   132→}

   138→function useWebRTC() {
   139→  const socketRef = useRef<Socket | null>(null)
   140→  const pcRef = useRef<RTCPeerConnection | null>(null)
   141→  const localStreamRef = useRef<MediaStream | null>(null)
   142→  const remoteStreamRef = useRef<MediaStream | null>(null)
   143→  const remotePeerIdRef = useRef<string | null>(null)
   144→
   145→  const [peers, setPeers] = useState<Peer[]>([])
   146→  const [myId, setMyId] = useState<string | null>(null)
   147→  const [myUsername, setMyUsername] = useState('')
   148→  const [isRegistered, setIsRegistered] = useState(false)
   149→  const [incomingCall, setIncomingCall] = useState<{ fromId: string; fromName: string; offer: RTCSessionDescriptionInit } | null>(null)
   150→  const [isInCall, setIsInCall] = useState(false)
   151→  const [remoteStream, setRemoteStream] = useState<MediaStream | null>(null)
   152→  const [localStream, setLocalStream] = useState<MediaStream | null>(null)
   153→  const [isMuted, setIsMuted] = useState(false)
   154→  const [isCameraOff, setIsCameraOff] = useState(false)
   155→  const [callStatus, setCallStatus] = useState<'idle' | 'ringing' | 'connecting' | 'connected' | 'ended'>('idle')
   156→
   157→  const myIdRef = useRef<string | null>(null)
   158→
   159→  const cleanupCall = useCallback(() => {
   160→    if (pcRef.current) { pcRef.current.close(); pcRef.current = null }
   161→    if (localStreamRef.current) { localStreamRef.current.getTracks().forEach((t) => t.stop()); localStreamRef.current = null }
   162→    remoteStreamRef.current = null
   163→    remotePeerIdRef.current = null
   164→    setRemoteStream(null); setLocalStream(null); setIsInCall(false)
   165→    setIsMuted(false); setIsCameraOff(false); setIncomingCall(null)
   166→  }, [])
   167→
   168→  useEffect(() => {
   169→    const socket = io('/?XTransformPort=3003', { transports: ['websocket', 'polling'] })
   170→    socketRef.current = socket
   171→
   172→    socket.on('registered', (data: { id: string; username: string }) => {
   173→      myIdRef.current = data.id
   174→      setMyId(data.id); setMyUsername(data.username); setIsRegistered(true)
   175→    })
   176→
   177→    socket.on('peer-list', (data: { peers: Peer[] }) => {
   178→      setPeers(data.peers.filter((p) => p.id !== myIdRef.current && !p.isAdmin))
   179→    })
   180→
   181→    socket.on('incoming-call', (data) => {
   182→      setIncomingCall(data); setCallStatus('ringing')
   183→    })
   184→
   185→    socket.on('call-answered', async (data) => {
   186→ const pc = pcRef.current
   187→      if (pc) { await pc.setRemoteDescription(new RTCSessionDescription(data.answer)); setCallStatus('connecting') }
   188→    })
   189→
   190→    socket.on('ice-candidate', async (data) => {
   191→      const pc = pcRef.current
   192→      if (pc && data.candidate) {
   193→        try { await pc.addIceCandidate(new RTCIceCandidate(data.candidate)) } catch (e) { /* ignore */ }
   194→      }
   195→    })
   196→
   197→    socket.on('call-rejected', () => { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) })
   198→    socket.on('call-ended', () => { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) })
   199→    socket.on('force-disconnected', () => { cleanupCall(); setCallStatus('ended'); setIsRegistered(false) })
   200→
   201→    return () => { socket.disconnect() }
   202→  }, [])
   203→
   204→  const getLocalStream = useCallback(async () => {
   205→    const stream = await navigator.mediaDevices.getUserMedia({ video: { width: 1280, height: 720, facingMode: 'user' }, audio: true })
   206→    localStreamRef.current = stream; setLocalStream(stream); return stream
   207→  }, [])
   208→
   209→  const createPeerConnection = useCallback((stream: MediaStream, remotePeerId: string) => {
   210→    const pc = new RTCPeerConnection({
   211→      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }, { urls: 'stun:stun1.l.google.com:19302' }]
   212→    })
   213→    pcRef.current = pc; remotePeerIdRef.current = remotePeerId
   214→    stream.getTracks().forEach((t) => pc.addTrack(t, stream))
   215→    const remoteMediaStream = new MediaStream()
   216→    remoteStreamRef.current = remoteMediaStream
   217→    pc.ontrack = (e) => { e.streams[0].getTracks().forEach((t) => remoteMediaStream.addTrack(t)); setRemoteStream(new MediaStream(remoteMediaStream.getTracks())) }
   218→    pc.onicecandidate = (e) => { if (e.candidate && socketRef.current) socketRef.current.emit('ice-candidate', { targetId: remotePeerId, candidate: e.candidate.toJSON() }) }
   219→    pc.onconnectionstatechange = () => {
   220→      if (pc.connectionState === 'connected') { setCallStatus('connected'); setIsInCall(true) }
   221→      else if (['disconnected', 'failed', 'closed'].includes(pc.connectionState)) { cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000) }
   222→    }
   223→    return pc
   224→  }, [cleanupCall, getLocalStream])
   225→
   226→  const register = useCallback((username: string) => { socketRef.current?.emit('register', { username, device: 'web-admin' }) }, [])
   227→
   228→  const callPeer = useCallback(async (targetId: string) => {
   229→    const stream = await getLocalStream()
   230→    createPeerConnection(stream, targetId)
   231→    const pc = pcRef.current; if (!pc) return
   232→    const offer = await pc.createOffer(); await pc.setLocalDescription(offer)
   233→    socketRef.current?.emit('call-offer', { targetId, offer: offer.toJSON() }); setCallStatus('connecting')
   234→  }, [getLocalStream, createPeerConnection])
   235→
   236→  const acceptCall = useCallback(async () => {
   237→    if (!incomingCall) return
   238→    const stream = await getLocalStream()
   239→    createPeerConnection(stream, incomingCall.fromId)
   240→    const pc = pcRef.current; if (!pc) return
   241→    await pc.setRemoteDescription(new RTCSessionDescription(incomingCall.offer))
   242→    const answer = await pc.createAnswer(); await pc.setLocalDescription(answer)
   243→    socketRef.current?.emit('call-answer', { targetId: incomingCall.fromId, answer: answer.toJSON() })
   244→    setIncomingCall(null); setCallStatus('connecting')
   245→  }, [incomingCall, getLocalStream, createPeerConnection])
   246→
   247→  const rejectCall = useCallback(() => {
   248→    if (incomingCall && socketRef.current) socketRef.current.emit('call-rejected', { targetId: incomingCall.fromId })
   249→    setIncomingCall(null); setCallStatus('idle')
   250→  }, [incomingCall])
   251→
   252→  const endCall = useCallback(() => {
   253→    if (socketRef.current && remotePeerIdRef.current) socketRef.current.emit('call-ended', { targetId: remotePeerIdRef.current })
   254→    cleanupCall(); setCallStatus('ended'); setTimeout(() => setCallStatus('idle'), 2000)
   255→  }, [cleanupCall])
   256→
   257→  const toggleMute = useCallback(() => { localStreamRef.current?.getAudioTracks().forEach((t) => { t.enabled = !t.enabled }); setIsMuted((p) => !p) }, [])
   258→  const toggleCamera = useCallback(() => { localStreamRef.current?.getVideoTracks().forEach((t) => { t.enabled = !t.enabled }); setIsCameraOff((p) => !p) }, [])
   259→
   260→  return { peers, myId, myUsername, isRegistered, register, callPeer, incomingCall, acceptCall, rejectCall, isInCall, remoteStream, localStream, endCall, isMuted, isCameraOff, toggleMute, toggleCamera, callStatus, socketRef }
   261→}

   267→function AdminDashboard({ onLogout }: { onLogout: () => void }) {
   268→  const [snapshot, setSnapshot] = useState<AdminSnapshot | null>(null)
   269→  const [connecting, setConnecting] = useState(true)
   270→  const [now, setNow] = useState(Date.now())
   271→  const socketRef = useRef<Socket | null>(null)
   272→
   273→  // Live ticker — updates every second for call duration timers
   274→  useEffect(() => {
   275→    const interval = setInterval(() => setNow(Date.now()), 1000)
   276→    return () => clearInterval(interval)
   277→  }, [])
   278→
   279→  useEffect(() => {
   280→    const socket = io('/?XTransformPort=3003', { transports: ['websocket', 'polling'] })
   281→    socketRef.current = socket
   282→
   283→    socket.on('connect', () => {
   284→      setConnecting(false)
   285→      socket.emit('admin-register', { secret: 'admin2024', username: 'Admin' })
   286→    })
   287→
   288→    socket.on('admin-registered', () => {
   289→      console.log('Admin registered')
   290→    })
   291→
   292→    socket.on('admin-snapshot', (data: AdminSnapshot) => {
   293→      setSnapshot(data)
   294→    })
   295→
   296→    socket.on('admin-rejected', (data) => {
   297→      console.error('Admin rejected:', data.reason)
   298→      setConnecting(false)
   299→    })
   300→
   301→    socket.on('disconnect', () => { setConnecting(true) })
   302→
   303→    return () => { socket.disconnect() }
   304→  }, [])
   305→
   306→  const forceDisconnect = (targetId: string) => {
   307→    socketRef.current?.emit('admin-force-disconnect', { targetId })
   308→  }
   309→
   310→  const endCall = (targetId: string) => {
   311→    socketRef.current?.emit('admin-end-call', { targetId })
   312→  }
   313→
   314→  const banUser = (targetId: string) => {
   315→    socketRef.current?.emit('admin-ban', { targetId })
   316→  }
   317→
   318→  const unbanUser = (username: string) => {
   319→    socketRef.current?.emit('admin-unban', { username })
   320→  }
   321→
   322→  const formatUptime = (ms: number) => {
   323→    const s = Math.floor(ms / 1000)
   324→    if (s < 60) return `${s}s`
   325→    const m = Math.floor(s / 60)
   326→    if (m < 60) return `${m}m ${s % 60}s`
   327→    const h = Math.floor(m / 60)
   328→    return `${h}h ${m % 60}m`
   329→  }
   330→
   331→  const activeCallDuration = (startedAt: number) => {
   332→    const diff = now - startedAt
   333→    const m = Math.floor(diff / 60000)
   334→    const s = Math.floor((diff % 60000) / 1000)
   335→    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   336→  }
   337→
   338→  const activeCallers = new Set(snapshot?.activeCalls.flatMap((c) => [c.callerId, c.calleeId]) ?? [])
   339→  const regularPeers = snapshot?.peers.filter((p) => !p.isAdmin) ?? []
   340→
   341→  return (
   342→    <div className="min-h-screen bg-gradient-to-br from-neutral-950 via-neutral-900 to-neutral-950">
   343→      <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   344→        {/* Header */}
   345→        <header className="flex items-center justify-between">
   346→          <div className="flex items-center gap-3">
   347→            <div className="w-10 h-10 rounded-xl bg-emerald-500/10 flex items-center justify-center">
   348→              <Shield className="w-5 h-5 text-emerald-400" />
   349→            </div>
   350→            <div>
   351→              <h1 className="text-lg font-bold text-white tracking-tight">Admin Dashboard</h1>
   352→              <p className="text-xs text-neutral-500">VideoCall control center</p>
   353→            </div>
   354→          </div>
   355→          <Button variant="ghost" onClick={onLogout} className="text-neutral-400 hover:text-red-400 gap-2">
   356→            <LogOut className="w-4 h-4" />
   357→            <span className="text-sm">Logout</span>
   358→          </Button>
   359→        </header>
   360→
   361→        {connecting && !snapshot && (
   362→          <Card className="bg-neutral-900/60 border-neutral-800">
   363→            <CardContent className="p-12 text-center">
   364→              <div className="w-10 h-10 border-3 border-emerald-400 border-t-transparent rounded-full animate-spin mx-auto mb-4" />
   365→              <p className="text-neutral-400">Connecting to server...</p>
   366→            </CardContent>
   367→          </Card>
   368→        )}
   369→
   370→        {snapshot && (
   371→          <>
   372→            {/* Stats row */}
   373→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   374→              <StatCard icon={<Users className="w-4 h-4" />} label="Online Users" value={snapshot.totalConnected} color="emerald" />
   375→              <StatCard icon={<PhoneCall className="w-4 h-4" />} label="Active Calls" value={snapshot.activeCalls.length} color="blue" />
   376→              <StatCard icon={<Smartphone className="w-4 h-4" />} label="Android" value={regularPeers.filter((p) => p.device?.includes('android')).length} color="violet" />
   377→              <StatCard icon={<Ban className="w-4 h-4" />} label="Banned" value={snapshot.bannedCount} color="red" />
   378→            </div>
   379→
   380→            {/* Active Calls */}
   381→            {snapshot.activeCalls.length > 0 && (
   382→              <div>
   383→                <h2 className="text-sm font-medium text-neutral-400 mb-3 flex items-center gap-2">
   384→                  <PhoneCall className="w-4 h-4" />
   385→                  Active Calls ({snapshot.activeCalls.length})
   386→                </h2>
   387→                <div className="grid gap-3">
   388→                  {snapshot.activeCalls.map((call) => (
   389→                    <Card key={call.id} className="bg-neutral-900/60 border-emerald-500/20">
   390→                      <CardContent className="p-4">
   391→                        <div className="flex items-center justify-between">
   392→                          <div className="flex items-center gap-3">
   393→                            <div className="flex flex-col items-center gap-1">
   394→                              <div className="w-8 h-8 rounded-full bg-emerald-500/20 flex items-center justify-center text-emerald-400 text-xs font-bold">
   395→                                {call.callerName.charAt(0).toUpperCase()}
   396→                              </div>
   397→                            </div>
   398→                            <div className="flex items-center gap-2">
   399→                              <span className="text-white text-sm font-medium">{call.callerName}</span>
   400→                              <div className="w-6 h-px bg-emerald-500" />
   401→                              <span className="text-white text-sm font-medium">{call.calleeName}</span>
   402→                            </div>
   403→                          </div>
   404→                          <div className="flex items-center gap-3">
   405→                            <div className="flex items-center gap-1.5 text-emerald-400 text-xs font-mono">
   406→                              <Clock className="w-3 h-3" />
   407→                              {activeCallDuration(call.startedAt)}
   408→                            </div>
   409→                            <Button
   410→                              size="sm"
   411→                              variant="ghost"
   412→                              onClick={() => endCall(call.callerId)}

   424→              </div>
   425→            )}
   426→
   427→            {/* Users list */}
   428→            <div>

   430→                <Users className="w-4 h-4" />
